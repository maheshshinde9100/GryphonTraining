        Why We Use Window FunctionsBasic Syntax:
window_function (expression) OVER (PARTITION BY column ORDER BY column)

Why We Use Window Functions?
- To compare the current row with previous/next rows.
- To calculate moving averages.
- To avoid complex subqueries.

Types of Window Functions:
1. ROW_NUMBER()
2. RANK()
3. DENSE_RANK()
4. SUM()
5. AVG()
6. LAG()
7. LEAD()

##table 'employees' with columns: id, name, salary, department, hire_date

1. ROW_NUMBER():
   SELECT name, salary, ROW_NUMBER() OVER (ORDER BY salary DESC) AS rn FROM employees;

2. RANK():
   SELECT name, salary, RANK() OVER (ORDER BY salary DESC) AS rnk FROM employees;

3. DENSE_RANK():
   SELECT name, salary, DENSE_RANK() OVER (ORDER BY salary DESC) AS drnk FROM employees;

4. SUM():
   SELECT name, salary, SUM(salary) OVER (PARTITION BY department ORDER BY salary) AS running_sum FROM employees;

5. AVG():
   SELECT name, salary, AVG(salary) OVER (PARTITION BY department) AS dept_avg FROM employees;

6. LAG():
   SELECT name, salary, LAG(salary) OVER (ORDER BY salary) AS prev_salary FROM employees;

7. LEAD():
   SELECT name, salary, LEAD(salary) OVER (ORDER BY salary) AS next_salary FROM employees;

8. Identify the employee who is the second last joiner:

   SELECT name, hire_date
   FROM (
       SELECT name, hire_date, ROW_NUMBER() OVER (ORDER BY hire_date DESC) AS rn
       FROM employees
   ) AS sub
   WHERE rn = 2;

7] find employees whose salary is greater than both previous and next 


SELECT * 
FROM (
    SELECT 
        name, 
        salary, 
        LAG(salary) OVER (ORDER BY salary) AS prev, 
        LEAD(salary) OVER (ORDER BY salary) AS next 
    FROM employees
) AS subquery_alias -- Mandatory alias for the subquery
WHERE salary > prev AND salary > next;


8] department wise salry jump > 10%


mysql> WITH SalaryHistory AS (
    ->     SELECT
    ->         id,
    ->         name,
    ->         department,
    ->         salary,
    ->         LAG(salary) OVER (
    ->             PARTITION BY department
    ->             ORDER BY salary ASC
    ->         ) AS previous_salary
    ->     FROM employees
    -> )
    -> SELECT
    ->     id,
    ->     name,
    ->     department,
    ->     salary AS current_salary,
    ->     previous_salary,
    ->     ROUND(((salary - previous_salary) / previous_salary) * 100, 2) AS jump_percentage
    -> FROM SalaryHistory
    -> WHERE (salary - previous_salary) / previous_salary > 0.10;
